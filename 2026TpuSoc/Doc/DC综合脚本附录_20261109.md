```tcl
# Hold recovery policy, factored out for selection/regression tests.
proc InSram {Name} {return [string match {Tpu/Core/Sram/*} $Name]}
proc SramAncestor {Name} {return [expr {$Name in {Tpu Tpu/Core Tpu/Core/Sram}}]}
proc TimingSignature {File} {
    set F [open $File r]
    set Text [read $F]
    close $F
    set Lines {}
    foreach Line [split $Text "\n"] {
        set Line [string trim $Line]
        if {$Line eq "" || [string match {#*} $Line]} {continue}
        # These are optimization permissions, not timing requirements.
        if {[regexp {^(set_fix_hold|set_dont_touch)\s} $Line]} {continue}
        lappend Lines $Line
    }
    return [lsort $Lines]
}
proc OutsideReferences {} {
    set Result [dict create]
    foreach_in_collection Cell [get_cells -hierarchical -filter {is_hierarchical == false}] {
        set Name [get_object_name $Cell]
        if {![InSram $Name]} {dict set Result $Name [get_attribute $Cell ref_name]}
    }
    return $Result
}
proc AssertSameReferences {Before After} {
    if {[dict size $Before] != [dict size $After]} {error "Outside-SRAM leaf count changed"}
    dict for {Name Ref} $Before {
        if {![dict exists $After $Name] || [dict get $After $Name] ne $Ref} {
            error "Outside-SRAM mapped cell changed: $Name"
        }
    }
}
```

[原始代码文件](DCScripts_20261109/Accelerator/AsicCommon.tcl)；SHA256：`87ff64e80cb4ccc333ef0371681c58e5419664b50810be3243aa2d7e27765893`。

```tcl
# Shared DC utilities. Timing paths are collections, not scalar values.
proc NumericMinimum {Values} {
    if {![llength $Values]} {error "No numeric timing values"}
    set Worst ""
    foreach Value $Values {
        if {![string is double -strict $Value] || [regexp -nocase {nan|inf} $Value]} {
            error "Invalid/non-finite timing value: $Value"
        }
        if {$Worst eq "" || $Value < $Worst} {set Worst $Value}
    }
    return $Worst
}
proc PathSlack {Kind args} {
    set Paths [eval get_timing_paths -delay_type $Kind -max_paths 1 $args]
    if {![sizeof_collection $Paths]} {error "No $Kind constrained timing path: $args"}
    set Values {}
    foreach_in_collection Path $Paths {lappend Values [get_attribute $Path slack]}
    return [NumericMinimum $Values]
}
proc FindCells {Pattern} {
    set Result ""
    foreach_in_collection Cell [get_cells -hierarchical -filter {is_hierarchical == false}] {
        if {[regexp $Pattern [get_object_name $Cell]]} {set Result [add_to_collection $Result $Cell]}
    }
    return $Result
}
proc RegisterPins {Cells Kind} {
    if {$Kind eq "clock"} {
        set Pattern {/(CK|CLK|CP|C)$}
    } elseif {$Kind eq "data"} {
        set Pattern {/D$}
    } else {error "Unknown register pin kind: $Kind"}
    set Result ""
    foreach_in_collection Cell $Cells {
        set Matches ""
        foreach_in_collection Pin [get_pins -of_objects $Cell] {
            if {[regexp $Pattern [get_object_name $Pin]]} {set Matches [add_to_collection $Matches $Pin]}
        }
        if {[sizeof_collection $Matches] != 1} {
            error "Expected one $Kind pin on [get_object_name $Cell]"
        }
        set Result [add_to_collection $Result $Matches]
    }
    if {![sizeof_collection $Result]} {error "No $Kind register pins selected"}
    return $Result
}
proc LibraryLeafArea {} {
    set Total 0.0
    array set Cache {}
    foreach_in_collection Cell [get_cells -hierarchical -filter {is_hierarchical == false}] {
        set Ref [get_attribute $Cell ref_name]
        if {$Ref in {**logic_0** **logic_1**}} {continue}
        if {![info exists Cache($Ref)]} {
            set LibCell [get_lib_cells -quiet */$Ref]
            if {[sizeof_collection $LibCell] != 1} {error "Unmapped or ambiguous reference: $Ref"}
            set Value [get_attribute $LibCell area]
            if {![string is double -strict $Value] || $Value < 0} {error "Invalid library area for $Ref"}
            set Cache($Ref) $Value
        }
        set Total [expr {$Total + $Cache($Ref)}]
    }
    if {$Total <= 0} {error "No positive mapped-cell area"}
    return $Total
}
proc FrozenBudgetSignature {File} {
    set F [open $File r]
    set Text [read $F]
    close $F
    set Result {}
    foreach Line [split $Text "\n"] {
        set Line [string trim $Line]
        if {[regexp {^(create_clock|create_generated_clock|set_clock_uncertainty|set_clock_transition|set_clock_latency|set_input_delay|set_output_delay|set_input_transition|set_load)\s} $Line]} {
            lappend Result $Line
        }
    }
    return [lsort $Result]
}
```

# Netlist-to-netlist comparison of the exact baseline and hold candidate.
# No RTL interpretation filters, SVF, reset assumptions or compare-point waivers.
proc Record {Key Value} {
    set F [open EquivStatus.tsv a]
    puts $F "$Key\t$Value"
    close $F
    puts "EQUIV_STAGE: $Key $Value"
}
proc RequireTrue {Key Value} {
    Record $Key $Value
    if {![string is boolean -strict $Value] || !$Value} {error "$Key did not return true"}
}
proc ReadStep {Script} {
    set Value [uplevel 1 $Script]
    if {[string is boolean -strict $Value] && !$Value} {error "Read/setup command returned false: $Script"}
}
proc Report {Name args} {
    if {[catch {redirect "${Name}.rpt" [linsert $args 0 $Name]} Why]} {
        Record REPORT_ERROR "$Name $Why"
    }
}
if {[catch {
    if {![info exists synopsys_auto_setup]} {error "Expected setup control unavailable"}

# Fresh CPU/ICB/shared-TCM mapping around the byte-identical frozen accelerator.
proc FrozenAccelerator {} {
    set Prefix u_e203_cpu/u_e203_nice_core/
    set Result [dict create]
    foreach_in_collection Cell [get_cells -hierarchical -filter {is_hierarchical == false}] {
        set Name [get_object_name $Cell]
        if {[string first $Prefix $Name]!=0} {continue}
        set Ref [get_attribute $Cell ref_name]
        if {$Ref in {**logic_0** **logic_1**}} {continue}
        dict set Result [string range $Name [string length $Prefix] end] $Ref
    }
    if {[dict size $Result]<1000} {error "Incomplete frozen accelerator signature"}
    return $Result
}

# Disjoint physical-library leaf areas; do not sum inclusive hierarchy reports.
proc AreaClass {Name Ref} {
    if {[string first "u_e203_cpu/u_e203_nice_core/" $Name]==0} {
        if {$Ref eq "RAMSP1024X16"} {return ACCELERATOR_PRIVATE_SRAM}
        if {$Ref eq "RAMSP4096X16"} {error "Unexpected shared macro inside accelerator"}
        return ACCELERATOR_LOGIC_AND_INTERFACES
    }
    if {$Ref eq "RAMSP1024X16"} {error "Private SRAM outside frozen accelerator"}
    if {$Ref eq "RAMSP4096X16"} {
        if {[string first "u_e203_srams/u_e203_itcm_ram/" $Name]==0} {return SHARED_ITCM_MACROS}
        if {[string first "u_e203_srams/u_e203_dtcm_ram/" $Name]==0} {return SHARED_DTCM_MACROS}
        error "Shared macro outside expected TCM boundaries"
    }
    if {[string first "u_e203_cpu/" $Name]==0} {return REUSED_CPU_AND_INTERNAL_ICB_LOGIC}
    if {[string first "u_e203_srams/" $Name]==0} {return SHARED_TCM_BINDING_LOGIC}
    return OTHER_SYSTEM_LOGIC
}
proc WriteAreaBreakdown {Out} {
    set Keys {ACCELERATOR_LOGIC_AND_INTERFACES ACCELERATOR_PRIVATE_SRAM REUSED_CPU_AND_INTERNAL_ICB_LOGIC SHARED_ITCM_MACROS SHARED_DTCM_MACROS SHARED_TCM_BINDING_LOGIC OTHER_SYSTEM_LOGIC}
    foreach Key $Keys {set Area($Key) 0.0; set Count($Key) 0}
    set Total 0.0
    array set Cache {}
    foreach_in_collection Cell [get_cells -hierarchical -filter {is_hierarchical == false}] {
        set Ref [get_attribute $Cell ref_name]
        if {$Ref in {**logic_0** **logic_1**}} {continue}
        set Key [AreaClass [get_object_name $Cell] $Ref]
        if {![info exists Cache($Ref)]} {
            set LibCell [get_lib_cells -quiet */$Ref]
            if {[sizeof_collection $LibCell]!=1} {error "Ambiguous/missing area $Ref"}
            set Value [get_attribute $LibCell area]
            if {![string is double -strict $Value] || !($Value>=0 && $Value<1e12)} {error "Invalid area $Ref"}
            set Cache($Ref) $Value
        }
        set Value $Cache($Ref)
        set Area($Key) [expr {$Area($Key)+$Value}]
        incr Count($Key)
        set Total [expr {$Total+$Value}]
    }
    if {$Count(ACCELERATOR_PRIVATE_SRAM)!=56 || $Count(SHARED_ITCM_MACROS)!=32 || $Count(SHARED_DTCM_MACROS)!=32} {
        error "Area inventory must contain private56/ITCM32/DTCM32 macros"
    }
    set Budget [expr {$Area(ACCELERATOR_LOGIC_AND_INTERFACES)+$Area(ACCELERATOR_PRIVATE_SRAM)}]
    if {$Total<=0 || $Budget<=0} {error "Empty area accounting"}
    set F [open [file join $Out area_breakdown.tsv] w]
    puts $F "component\tleaf_count\tarea_um2\tarea_mm2\tin_accelerator_budget"
    foreach Key $Keys {
        set Included [expr {[string first "ACCELERATOR_" $Key]==0}]
        puts $F "$Key\t$Count($Key)\t[format %.9f $Area($Key)]\t[format %.12f [expr {$Area($Key)/1e6}]]\t$Included"
        puts "AREA_COMPONENT: $Key area_mm2=[format %.9f [expr {$Area($Key)/1e6}]] leaves=$Count($Key) budget=$Included"
    }
    close $F
    set F [open [file join $Out area_totals.txt] w]
    puts $F "SYSTEM_AREA_UM2: $Total"
    puts $F "ACCELERATOR_BUDGET_AREA_UM2: $Budget"
    puts $F "SHARED_CAPACITY_BYTES: 524288"
    puts $F "SHARED_ITCM_BYTES: 262144"
    puts $F "SHARED_DTCM_BYTES: 262144"
    puts $F "PRIVATE_CAPACITY_BYTES: 114688"
    puts $F "BOUNDARY: e203_cpu_top; external fixed peripherals excluded; SRAM bank/mux binding logic listed separately"
    close $F
}

// Candidate physical implementation of the competition TCM RAM interface.
// CPU/LSU/TCM controllers are unchanged. Read results are valid after read edges.
module sirv_gnrl_ram #(
    parameter DP=32768, DW=64, FORCE_X2ZERO=0, MW=8, AW=15
) (
    input sd,ds,ls,rst_n,clk,cs,we,
    input [AW-1:0] addr,
    input [DW-1:0] din,
    input [MW-1:0] wem,
    output [DW-1:0] dout
);
    localparam BANKS=DP/4096;
    localparam LANES=DW/16;
    localparam SEL=AW-12;
    wire [DW-1:0] BankData[0:BANKS-1];
    reg [SEL-1:0] ReadBank;
    always @(posedge clk) if (cs && !we) ReadBank<=addr[AW-1:12];
    assign dout=BankData[ReadBank];
    for (genvar Bank=0;Bank<BANKS;Bank++) begin: Banks
        wire Enable=cs && addr[AW-1:12]==SEL'(Bank);
        for (genvar Lane=0;Lane<LANES;Lane++) begin: Lanes
            wire [15:0] WriteMask={{8{~wem[2*Lane+1]}},{8{~wem[2*Lane]}}};
            RAMSP4096X16 Macro (
                .Q(BankData[Bank][16*Lane+:16]), .CLK(clk), .CEN(~Enable),
                .GWEN(~we), .WEN(WriteMask), .A(addr[11:0]), .D(din[16*Lane+:16]),
                .EMA(3'd2), .EMAW(2'd0), .RET1N(1'b1)
            );
        end
    end
    // No RAM initialization, reset-to-zero, or simulation X replacement.
    // Sleep pins are unused by the original functional RAM and remain unused.
    // synopsys translate_off
    initial begin
        if (!((DW==64 && MW==8 && AW==15 && DP==32768) ||
              (DW==32 && MW==4 && AW==16 && DP==65536)))
            $fatal(1,"UNSUPPORTED_COMPETITION_TCM_SHAPE DW=%0d MW=%0d AW=%0d DP=%0d",DW,MW,AW,DP);
    end
    // synopsys translate_on
endmodule
